# Anrufen Online — C++ istemcisi (yerel sürüm). Belge: docs/09-cpp.md
#   cmake -S native -B native/build -G Ninja && cmake --build native/build && ctest --test-dir native/build
cmake_minimum_required(VERSION 3.16)
project(anrufen_native CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(MSVC)
  add_compile_options(/W4 /utf-8)
else()
  add_compile_options(-Wall -Wextra)
endif()

find_package(ZLIB REQUIRED)

# Yayın anahtarı: native/anahtar.txt (64 onaltılık karakter, git dışı). Yoksa geliştirme anahtarı (herkese açık).
set(ANAHTAR_DOSYASI "${CMAKE_CURRENT_SOURCE_DIR}/anahtar.txt")
set(GODAT_KEY_DEF "")
if(EXISTS "${ANAHTAR_DOSYASI}")
  file(READ "${ANAHTAR_DOSYASI}" _k)
  string(STRIP "${_k}" _k)
  if(NOT _k MATCHES "^[0-9a-fA-F]+$")
    message(FATAL_ERROR "anahtar.txt yalnız onaltılık karakter içermeli")
  endif()
  string(LENGTH "${_k}" _n)
  if(NOT _n EQUAL 64)
    message(FATAL_ERROR "anahtar.txt 64 onaltılık karakter (32 bayt) olmalı")
  endif()
  string(REGEX REPLACE "([0-9a-fA-F][0-9a-fA-F])" "0x\\1," _bytes "${_k}")
  string(REGEX REPLACE ",$" "" _bytes "${_bytes}")
  set(GODAT_KEY_DEF "GODAT_RELEASE_KEY=${_bytes}")
  message(STATUS "godat: yayın anahtarı kullanılıyor")
else()
  message(STATUS "godat: geliştirme anahtarı (native/anahtar.txt yok)")
endif()

add_library(godat STATIC src/godat/godat.cpp src/godat/chacha20.cpp)
target_include_directories(godat PUBLIC src)
target_link_libraries(godat PUBLIC ZLIB::ZLIB)
if(GODAT_KEY_DEF)
  target_compile_definitions(godat PRIVATE "${GODAT_KEY_DEF}")
endif()

add_executable(godat_dump src/tools/godat_dump.cpp)
target_link_libraries(godat_dump PRIVATE godat)
add_executable(godat_pack src/tools/godat_pack.cpp)
target_link_libraries(godat_pack PRIVATE godat)

enable_testing()
add_executable(test_godat tests/test_godat.cpp)
target_link_libraries(test_godat PRIVATE godat)
add_test(NAME godat COMMAND test_godat)
