# Anrufen Online — C++ istemcisi (yerel sürüm). Belge: docs/09-cpp.md
#   cmake -S native -B native/build -G Ninja && cmake --build native/build && ctest --test-dir native/build
cmake_minimum_required(VERSION 3.16)
project(anrufen_native CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# uyarılar yalnız kendi kodumuza (SDL'e değil)
if(MSVC)
  set(ANRUFEN_WARN /W4 /utf-8)
else()
  set(ANRUFEN_WARN -Wall -Wextra)
endif()

# zlib: sistemde varsa o (Linux), yoksa depodaki kopya (third_party/zlib: Windows, Android, iOS)
option(ANRUFEN_YEREL_ZLIB "Depodaki zlib kopyasını kullan" OFF)
if(NOT ANRUFEN_YEREL_ZLIB)
  find_package(ZLIB QUIET)
endif()
if(NOT ZLIB_FOUND)
  enable_language(C)
  file(GLOB ZLIB_SRC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/zlib/*.c)
  add_library(zlib_yerel STATIC ${ZLIB_SRC})
  target_include_directories(zlib_yerel PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/zlib)
  add_library(ZLIB::ZLIB ALIAS zlib_yerel)
  message(STATUS "zlib: depodaki kopya (third_party/zlib)")
endif()

# Yayın anahtarı: native/anahtar.txt (64 onaltılık karakter, git dışı). Yoksa geliştirme anahtarı (herkese açık).
set(ANAHTAR_DOSYASI "${CMAKE_CURRENT_SOURCE_DIR}/anahtar.txt")
set(GODAT_KEY_DEF "")
if(EXISTS "${ANAHTAR_DOSYASI}")
  file(READ "${ANAHTAR_DOSYASI}" _k)
  string(STRIP "${_k}" _k)
  if(NOT _k MATCHES "^[0-9a-fA-F]+$")
    message(FATAL_ERROR "anahtar.txt yalnız onaltılık karakter içermeli")
  endif()
  string(LENGTH "${_k}" _n)
  if(NOT _n EQUAL 64)
    message(FATAL_ERROR "anahtar.txt 64 onaltılık karakter (32 bayt) olmalı")
  endif()
  string(REGEX REPLACE "([0-9a-fA-F][0-9a-fA-F])" "0x\\1," _bytes "${_k}")
  string(REGEX REPLACE ",$" "" _bytes "${_bytes}")
  set(GODAT_KEY_DEF "GODAT_RELEASE_KEY=${_bytes}")
  message(STATUS "godat: yayın anahtarı kullanılıyor")
else()
  message(STATUS "godat: geliştirme anahtarı (native/anahtar.txt yok)")
endif()

add_library(godat STATIC src/godat/godat.cpp src/godat/chacha20.cpp)
target_include_directories(godat PUBLIC src)
target_link_libraries(godat PUBLIC ZLIB::ZLIB)
if(GODAT_KEY_DEF)
  target_compile_definitions(godat PRIVATE "${GODAT_KEY_DEF}")
endif()

add_executable(godat_dump src/tools/godat_dump.cpp)
target_link_libraries(godat_dump PRIVATE godat)
add_executable(godat_pack src/tools/godat_pack.cpp)
target_link_libraries(godat_pack PRIVATE godat)

# Motor (pencereden bağımsız): PNG/JSON, sanal dosya sistemi, tuval, varlıklar, harita, dünya
add_library(motor STATIC src/core/png.cpp src/core/json.cpp src/engine/vfs.cpp src/engine/canvas.cpp
            src/engine/assets.cpp src/engine/map.cpp src/engine/balance.cpp src/engine/world.cpp)
target_include_directories(motor PUBLIC src)
target_link_libraries(motor PUBLIC godat)

# Ekransız test modu (SDL gerekmez): senaryo → ekran görüntüsü, durum
add_executable(anrufen_test src/app/headless.cpp)
target_link_libraries(anrufen_test PRIVATE motor)

# Pencereli oyun: SDL2 kaynağı depoda (third_party/SDL2, sürüm ve kaynak: third_party/README.md)
option(ANRUFEN_SDL "SDL2'yi derle ve pencereli oyunu kur" ON)
if(ANRUFEN_SDL)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/SDL2 EXCLUDE_FROM_ALL)
  add_executable(anrufen WIN32 src/app/main.cpp)
  target_link_libraries(anrufen PRIVATE motor SDL2::SDL2main SDL2::SDL2-static)
endif()

# Oyun veri paketi: www/assets → build/veri/oyun.gdp (şifreli; cmake --build … --target veri)
add_custom_target(veri
  COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_BINARY_DIR}/veri
  COMMAND godat_pack -o ${CMAKE_BINARY_DIR}/veri/oyun.gdp --dizin ${CMAKE_CURRENT_SOURCE_DIR}/../www/assets
  DEPENDS godat_pack COMMENT "veri paketi: www/assets → veri/oyun.gdp")

foreach(t godat godat_dump godat_pack motor anrufen_test)
  target_compile_options(${t} PRIVATE ${ANRUFEN_WARN})
endforeach()
if(TARGET anrufen)
  target_compile_options(anrufen PRIVATE ${ANRUFEN_WARN})
endif()

enable_testing()
add_executable(test_godat tests/test_godat.cpp)
target_link_libraries(test_godat PRIVATE godat)
add_test(NAME godat COMMAND test_godat)
add_executable(test_motor tests/test_motor.cpp)
target_link_libraries(test_motor PRIVATE motor)
add_test(NAME motor COMMAND test_motor ${CMAKE_CURRENT_SOURCE_DIR}/../www/assets)
